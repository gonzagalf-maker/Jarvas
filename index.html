function montarTextoPagina(itens) { const preparados = (itens || []) .filter(item => item && typeof item.str === 'string' && item.str.length) .map((item, ordem) => { const t = item.transform || [1, 0, 0, 1, 0, 0]; return { texto: item.str.replace(/[\t\r\n]+/g, ' '), x: Number(t[4]) || 0, y: Number(t[5]) || 0, largura: Math.max(0, Number(item.width) || 0), fonte: Math.max( 1, Math.hypot(Number(t[2]) || 0, Number(t[3]) || 0), Math.hypot(Number(t[0]) || 0, Number(t[1]) || 0) ), ordem }; }); // Ordena aproximadamente de cima para baixo e da esquerda para a direita. preparados.sort((a, b) => b.y - a.y || a.x - b.x || a.ordem - b.ordem ); // Agrupa os trechos que pertencem à mesma linha. const linhas = []; for (const item of preparados) { let melhorLinha = null; let menorDiferenca = Infinity; for (const linha of linhas) { const tolerancia = Math.max( 2, Math.min(linha.fonte, item.fonte) * 0.28 ); const diferenca = Math.abs(linha.y - item.y); if ( diferenca <= tolerancia && diferenca < menorDiferenca ) { melhorLinha = linha; menorDiferenca = diferenca; } } if (!melhorLinha) { melhorLinha = { y: item.y, fonte: item.fonte, itens: [] }; linhas.push(melhorLinha); } melhorLinha.itens.push(item); melhorLinha.y = melhorLinha.itens.reduce((soma, i) => soma + i.y, 0) / melhorLinha.itens.length; melhorLinha.fonte = melhorLinha.itens.reduce((soma, i) => soma + i.fonte, 0) / melhorLinha.itens.length; } // Ordena as linhas e os trechos dentro de cada linha. linhas.sort((a, b) => b.y - a.y); const resultado = []; for (const linha of linhas) { linha.itens.sort((a, b) => a.x - b.x || a.ordem - b.ordem ); let textoLinha = ''; let anterior = null; for (const item of linha.itens) { const atual = item.texto; if (anterior) { const fimAnterior = anterior.x + anterior.largura; const distancia = item.x - fimAnterior; const fonte = Math.max(anterior.fonte, item.fonte); const comecaPontuacao = /^[,.;:!?%)\]}]/.test(atual.trim()); const terminaAbertura = /[(\[{]$/.test(textoLinha.trimEnd()); if ( distancia > fonte * 0.2 && !comecaPontuacao && !terminaAbertura && !/\s$/.test(textoLinha) ) { textoLinha += ' '; } } textoLinha += atual; anterior = item; } textoLinha = textoLinha.replace(/[ \t]+/g, ' ').trim(); if (textoLinha) { resultado.push(textoLinha); } } return resultado.join('\n'); }
