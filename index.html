function reconstruirTextoPDF(conteudo){ const itens = conteudo.items .filter(item => item.str && item.str.trim()) .map(item => ({ texto: item.str, x: item.transform[4], y: item.transform[5], largura: item.width || 0, fonte: Math.max( 1, Math.hypot(item.transform[0], item.transform[1]) ), fimLinha: Boolean(item.hasEOL) })) .sort((a, b) => { // Primeiro, ordenar de cima para baixo. const diferencaY = b.y - a.y; if(Math.abs(diferencaY) > 2) return diferencaY; // Na mesma linha, ordenar da esquerda para a direita. return a.x - b.x; }); const linhas = []; for(const item of itens){ let linha = linhas.find(l => Math.abs(l.y - item.y) <= Math.max(2, Math.min(l.fonte, item.fonte) * 0.25) ); if(!linha){ linha = { y: item.y, fonte: item.fonte, itens: [] }; linhas.push(linha); } linha.itens.push(item); } linhas.sort((a, b) => b.y - a.y); const resultado = []; for(const linha of linhas){ linha.itens.sort((a, b) => a.x - b.x); let textoLinha = ''; let anterior = null; for(const item of linha.itens){ const atual = item.texto.trim(); if(!atual) continue; if(anterior){ const fimAnterior = anterior.x + anterior.largura; const distancia = item.x - fimAnterior; const tamanhoFonte = Math.max( anterior.fonte, item.fonte ); // Distância grande pode indicar outra coluna. if(distancia > tamanhoFonte * 2){ textoLinha += '\n'; }else if( distancia > tamanhoFonte * 0.22 && !/\s$/.test(textoLinha) && !/^[,.;:!?%)\]}]/.test(atual) && !/[(\[{]$/.test(textoLinha) ){ textoLinha += ' '; } } textoLinha += atual; anterior = item; if(item.fimLinha){ // A quebra definitiva será aplicada ao terminar a linha. } } if(textoLinha.trim()){ resultado.push(textoLinha.trim()); } } return resultado.join('\n'); }
